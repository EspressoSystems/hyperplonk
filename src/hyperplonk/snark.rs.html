32impl<E, PCS> HyperPlonkSNARK<E, PCS> for PolyIOP<E::ScalarField>
33where
34    E: Pairing,
35    // Ideally we want to access polynomial as PCS::Polynomial, instead of instantiating it here.
36    // But since PCS::Polynomial can be both univariate or multivariate in our implementation
37    // we cannot bound PCS::Polynomial with a property trait bound.
38    PCS: PolynomialCommitmentScheme<
39        E,
40        Polynomial = Arc<DenseMultilinearExtension<E::ScalarField>>,
41        Point = Vec<E::ScalarField>,
42        Evaluation = E::ScalarField,
43        Commitment = Commitment<E>,
44        BatchProof = BatchProof<E, PCS>,
45    >,
46{
47    type Index = HyperPlonkIndex<E::ScalarField>;
48    type ProvingKey = HyperPlonkProvingKey<E, PCS>;
49    type VerifyingKey = HyperPlonkVerifyingKey<E, PCS>;
50    type Proof = HyperPlonkProof<E, Self, PCS>;
51
52    fn preprocess(
53        index: &Self::Index,
54        pcs_srs: &PCS::SRS,
55    ) -> Result<(Self::ProvingKey, Self::VerifyingKey), HyperPlonkErrors> {
56        let num_vars = index.num_variables();
57        let supported_ml_degree = num_vars;
58
59        // extract PCS prover and verifier keys from SRS
60        let (pcs_prover_param, pcs_verifier_param) =
61            PCS::trim(pcs_srs, None, Some(supported_ml_degree))?;
62
63        // build permutation oracles
64        let mut permutation_oracles = vec![];
65        let mut perm_comms = vec![];
66        let chunk_size = 1 << num_vars;
67        for i in 0..index.num_witness_columns() {
68            let perm_oracle = Arc::new(DenseMultilinearExtension::from_evaluations_slice(
69                num_vars,
70                &index.permutation[i * chunk_size..(i + 1) * chunk_size],
71            ));
72            let perm_comm = PCS::commit(&pcs_prover_param, &perm_oracle)?;
73            permutation_oracles.push(perm_oracle);
74            perm_comms.push(perm_comm);
75        }
76
77        // build selector oracles and commit to it
78        let selector_oracles: Vec<Arc<DenseMultilinearExtension<E::ScalarField>>> = index
79            .selectors
80            .iter()
81            .map(|s| Arc::new(DenseMultilinearExtension::from(s)))
82            .collect();
83
84        let selector_commitments = selector_oracles
85            .par_iter()
86            .map(|poly| PCS::commit(&pcs_prover_param, poly))
87            .collect::<Result<Vec<_>, _>>()?;
88
89        Ok((
90            Self::ProvingKey {
91                params: index.params.clone(),
92                permutation_oracles,
93                selector_oracles,
94                selector_commitments: selector_commitments.clone(),
95                permutation_commitments: perm_comms.clone(),
96                pcs_param: pcs_prover_param,
97            },
98            Self::VerifyingKey {
99                params: index.params.clone(),
100                pcs_param: pcs_verifier_param,
101                selector_commitments,
102                perm_commitments: perm_comms,
103            },
104        ))
105    }
106
107    /// Generate HyperPlonk SNARK proof.
108    ///
109    /// Inputs:
110    /// - `pk`: circuit proving key
111    /// - `pub_input`: online public input of length 2^\ell
112    /// - `witness`: witness assignment of length 2^n
113    ///
114    /// Outputs:
115    /// - The HyperPlonk SNARK proof.
116    ///
117    /// Steps:
118    ///
119    /// 1. Commit Witness polynomials `w_i(x)` and append commitment to
120    ///    transcript
121    ///
122    /// 2. Run ZeroCheck on
123    ///
124    ///     `f(q_0(x),...q_l(x), w_0(x),...w_d(x))`  
125    ///
126    /// where `f` is the constraint polynomial i.e.,
127    /// ```ignore
128    ///     f(q_l, q_r, q_m, q_o, w_a, w_b, w_c)
129    ///     = q_l w_a(x) + q_r w_b(x) + q_m w_a(x)w_b(x) - q_o w_c(x)
130    /// ```
131    /// in vanilla plonk, and obtain a ZeroCheckSubClaim
132    ///
133    /// 3. Run permutation check on `\{w_i(x)\}` and `permutation_oracle`, and
134    ///    obtain a PermCheckSubClaim.
135    ///
136    /// 4. Generate evaluations and corresponding proofs
137    /// - 4.1. (deferred) batch opening prod(x) at
138    ///   - [0, perm_check_point]
139    ///   - [1, perm_check_point]
140    ///   - [perm_check_point, 0]
141    ///   - [perm_check_point, 1]
142    ///   - [1,...1, 0]
143    ///
144    /// - 4.2. permutation check evaluations and proofs
145    ///   - 4.2.1. (deferred) wi_poly(perm_check_point)
146    ///
147    /// - 4.3. zero check evaluations and proofs
148    ///   - 4.3.1. (deferred) wi_poly(zero_check_point)
149    ///   - 4.3.2. (deferred) selector_poly(zero_check_point)
150    ///
151    /// - 4.4. public input consistency checks
152    ///   - pi_poly(r_pi) where r_pi is sampled from transcript
153    ///
154    /// - 5. deferred batch opening
155    fn prove(
156        pk: &Self::ProvingKey,
157        pub_input: &[E::ScalarField],
158        witnesses: &[WitnessColumn<E::ScalarField>],
159    ) -> Result<Self::Proof, HyperPlonkErrors> {
160        let start = start_timer!(|| "hyperplonk proving");
161        let mut transcript = IOPTranscript::<E::ScalarField>::new(b"hyperplonk");
162
163        prover_sanity_check(&pk.params, pub_input, witnesses)?;
164
165        // witness assignment of length 2^n
166        let num_vars = pk.params.num_variables();
167
168        // online public input of length 2^\ell
169        let ell = log2(pk.params.num_pub_input) as usize;
170
171        // We use accumulators to store the polynomials and their eval points.
172        // They are batch opened at a later stage.
173        let mut pcs_acc = PcsAccumulator::<E, PCS>::new(num_vars);
174
175        // =======================================================================
176        // 1. Commit Witness polynomials `w_i(x)` and append commitment to
177        // transcript
178        // =======================================================================
179        let step = start_timer!(|| "commit witnesses");
180
181        let witness_polys: Vec<Arc<DenseMultilinearExtension<E::ScalarField>>> = witnesses
182            .iter()
183            .map(|w| Arc::new(DenseMultilinearExtension::from(w)))
184            .collect();
185
186        let witness_commits = witness_polys
187            .par_iter()
188            .map(|x| PCS::commit(&pk.pcs_param, x).unwrap())
189            .collect::<Vec<_>>();
190        for w_com in witness_commits.iter() {
191            transcript.append_serializable_element(b"w", w_com)?;
192        }
193
194        end_timer!(step);
195        // =======================================================================
196        // 2 Run ZeroCheck on
197        //
198        //     `f(q_0(x),...q_l(x), w_0(x),...w_d(x))`
199        //
200        // where `f` is the constraint polynomial i.e.,
201        //
202        //     f(q_l, q_r, q_m, q_o, w_a, w_b, w_c)
203        //     = q_l w_a(x) + q_r w_b(x) + q_m w_a(x)w_b(x) - q_o w_c(x)
204        //
205        // in vanilla plonk, and obtain a ZeroCheckSubClaim
206        // =======================================================================
207        let step = start_timer!(|| "ZeroCheck on f");
208
209        let fx = build_f(
210            &pk.params.gate_func,
211            pk.params.num_variables(),
212            &pk.selector_oracles,
213            &witness_polys,
214        )?;
215
216        let zero_check_proof = <Self as ZeroCheck<E::ScalarField>>::prove(&fx, &mut transcript)?;
217        end_timer!(step);
218        // =======================================================================
219        // 3. Run permutation check on `\{w_i(x)\}` and `permutation_oracle`, and
220        // obtain a PermCheckSubClaim.
221        // =======================================================================
222        let step = start_timer!(|| "Permutation check on w_i(x)");
223
224        let (perm_check_proof, prod_x, frac_poly) = <Self as PermutationCheck<E, PCS>>::prove(
225            &pk.pcs_param,
226            &witness_polys,
227            &witness_polys,
228            &pk.permutation_oracles,
229            &mut transcript,
230        )?;
231        let perm_check_point = &perm_check_proof.zero_check_proof.point;
232
233        end_timer!(step);
234        // =======================================================================
235        // 4. Generate evaluations and corresponding proofs
236        // - permcheck
237        //  1. (deferred) batch opening prod(x) at
238        //   - [perm_check_point]
239        //   - [perm_check_point[2..n], 0]
240        //   - [perm_check_point[2..n], 1]
241        //   - [1,...1, 0]
242        //  2. (deferred) batch opening frac(x) at
243        //   - [perm_check_point]
244        //   - [perm_check_point[2..n], 0]
245        //   - [perm_check_point[2..n], 1]
246        //  3. (deferred) batch opening s_id(x) at
247        //   - [perm_check_point]
248        //  4. (deferred) batch opening perms(x) at
249        //   - [perm_check_point]
250        //  5. (deferred) batch opening witness_i(x) at
251        //   - [perm_check_point]
252        //
253        // - zero check evaluations and proofs
254        //   - 4.3.1. (deferred) wi_poly(zero_check_point)
255        //   - 4.3.2. (deferred) selector_poly(zero_check_point)
256        //
257        // - 4.4. (deferred) public input consistency checks
258        //   - pi_poly(r_pi) where r_pi is sampled from transcript
259        // =======================================================================
260        let step = start_timer!(|| "opening and evaluations");
261
262        // (perm_check_point[2..n], 0)
263        let perm_check_point_0 = [
264            &[E::ScalarField::zero()],
265            &perm_check_point[0..num_vars - 1],
266        ]
267        .concat();
268        // (perm_check_point[2..n], 1)
269        let perm_check_point_1 =
270            [&[E::ScalarField::one()], &perm_check_point[0..num_vars - 1]].concat();
271        // (1, ..., 1, 0)
272        let prod_final_query_point = [
273            vec![E::ScalarField::zero()],
274            vec![E::ScalarField::one(); num_vars - 1],
275        ]
276        .concat();
277
278        // prod(x)'s points
279        pcs_acc.insert_poly_and_points(&prod_x, &perm_check_proof.prod_x_comm, perm_check_point);
280        pcs_acc.insert_poly_and_points(&prod_x, &perm_check_proof.prod_x_comm, &perm_check_point_0);
281        pcs_acc.insert_poly_and_points(&prod_x, &perm_check_proof.prod_x_comm, &perm_check_point_1);
282        pcs_acc.insert_poly_and_points(
283            &prod_x,
284            &perm_check_proof.prod_x_comm,
285            &prod_final_query_point,
286        );
287
288        // frac(x)'s points
289        pcs_acc.insert_poly_and_points(&frac_poly, &perm_check_proof.frac_comm, perm_check_point);
290        pcs_acc.insert_poly_and_points(
291            &frac_poly,
292            &perm_check_proof.frac_comm,
293            &perm_check_point_0,
294        );
295        pcs_acc.insert_poly_and_points(
296            &frac_poly,
297            &perm_check_proof.frac_comm,
298            &perm_check_point_1,
299        );
300
301        // perms(x)'s points
302        for (perm, pcom) in pk
303            .permutation_oracles
304            .iter()
305            .zip(pk.permutation_commitments.iter())
306        {
307            pcs_acc.insert_poly_and_points(perm, pcom, perm_check_point);
308        }
309
310        // witnesses' points
311        // TODO: refactor so it remains correct even if the order changed
312        for (wpoly, wcom) in witness_polys.iter().zip(witness_commits.iter()) {
313            pcs_acc.insert_poly_and_points(wpoly, wcom, perm_check_point);
314        }
315        for (wpoly, wcom) in witness_polys.iter().zip(witness_commits.iter()) {
316            pcs_acc.insert_poly_and_points(wpoly, wcom, &zero_check_proof.point);
317        }
318
319        //   - 4.3.2. (deferred) selector_poly(zero_check_point)
320        pk.selector_oracles
321            .iter()
322            .zip(pk.selector_commitments.iter())
323            .for_each(|(poly, com)| {
324                pcs_acc.insert_poly_and_points(poly, com, &zero_check_proof.point)
325            });
326
327        // - 4.4. public input consistency checks
328        //   - pi_poly(r_pi) where r_pi is sampled from transcript
329        let r_pi = transcript.get_and_append_challenge_vectors(b"r_pi", ell)?;
330        // padded with zeros
331        let r_pi_padded = [r_pi, vec![E::ScalarField::zero(); num_vars - ell]].concat();
332        // Evaluate witness_poly[0] at r_pi||0s which is equal to public_input evaluated
333        // at r_pi. Assumes that public_input is a power of 2
334        pcs_acc.insert_poly_and_points(&witness_polys[0], &witness_commits[0], &r_pi_padded);
335        end_timer!(step);
336
337        // =======================================================================
338        // 5. deferred batch opening
339        // =======================================================================
340        let step = start_timer!(|| "deferred batch openings prod(x)");
341        let batch_openings = pcs_acc.multi_open(&pk.pcs_param, &mut transcript)?;
342        end_timer!(step);
343
344        end_timer!(start);
345
346        Ok(HyperPlonkProof {
347            // PCS commit for witnesses
348            witness_commits,
349            // batch_openings,
350            batch_openings,
351            // =======================================================================
352            // IOP proofs
353            // =======================================================================
354            // the custom gate zerocheck proof
355            zero_check_proof,
356            // the permutation check proof for copy constraints
357            perm_check_proof,
358        })
359    }
360
361    /// Verify the HyperPlonk proof.
362    ///
363    /// Inputs:
364    /// - `vk`: verification key
365    /// - `pub_input`: online public input
366    /// - `proof`: HyperPlonk SNARK proof
367    ///
368    /// Outputs:
369    /// - Return a boolean on whether the verification is successful
370    ///
371    /// 1. Verify zero_check_proof on
372    ///
373    ///     `f(q_0(x),...q_l(x), w_0(x),...w_d(x))`
374    ///
375    /// where `f` is the constraint polynomial i.e.,
376    /// ```ignore
377    ///     f(q_l, q_r, q_m, q_o, w_a, w_b, w_c)
378    ///     = q_l w_a(x) + q_r w_b(x) + q_m w_a(x)w_b(x) - q_o w_c(x)
379    /// ```
380    /// in vanilla plonk, and obtain a ZeroCheckSubClaim
381    ///
382    /// 2. Verify perm_check_proof on `\{w_i(x)\}` and `permutation_oracles`
383    ///
384    /// 3. check subclaim validity
385    ///
386    /// 4. Verify the opening against the commitment:
387    /// - check permutation check evaluations
388    /// - check zero check evaluations
389    /// - public input consistency checks
390    fn verify(
391        vk: &Self::VerifyingKey,
392        pub_input: &[E::ScalarField],
393        proof: &Self::Proof,
394    ) -> Result<bool, HyperPlonkErrors> {
395        let start = start_timer!(|| "hyperplonk verification");
396
397        let mut transcript = IOPTranscript::<E::ScalarField>::new(b"hyperplonk");
398
399        let num_selectors = vk.params.num_selector_columns();
400        let num_witnesses = vk.params.num_witness_columns();
401        let num_vars = vk.params.num_variables();
402
403        //  online public input of length 2^\ell
404        let ell = log2(vk.params.num_pub_input) as usize;
405
406        // =======================================================================
407        // 0. sanity checks
408        // =======================================================================
409        // public input length
410        if pub_input.len() != vk.params.num_pub_input {
411            return Err(HyperPlonkErrors::InvalidProver(format!(
412                "Public input length is not correct: got {}, expect {}",
413                pub_input.len(),
414                1 << ell
415            )));
416        }
417
418        // Extract evaluations from openings
419        let prod_evals = &proof.batch_openings.f_i_eval_at_point_i[0..4];
420        let frac_evals = &proof.batch_openings.f_i_eval_at_point_i[4..7];
421        let perm_evals = &proof.batch_openings.f_i_eval_at_point_i[7..7 + num_witnesses];
422        let witness_perm_evals =
423            &proof.batch_openings.f_i_eval_at_point_i[7 + num_witnesses..7 + 2 * num_witnesses];
424        let witness_gate_evals =
425            &proof.batch_openings.f_i_eval_at_point_i[7 + 2 * num_witnesses..7 + 3 * num_witnesses];
426        let selector_evals = &proof.batch_openings.f_i_eval_at_point_i
427            [7 + 3 * num_witnesses..7 + 3 * num_witnesses + num_selectors];
428        let pi_eval = proof.batch_openings.f_i_eval_at_point_i.last().unwrap();
429
430        // =======================================================================
431        // 1. Verify zero_check_proof on `f(q_0(x),...q_l(x), w_0(x),...w_d(x))`
432        //
433        // where `f` is the constraint polynomial i.e.,
434        //
435        //     f(q_l, q_r, q_m, q_o, w_a, w_b, w_c)
436        //     = q_l w_a(x) + q_r w_b(x) + q_m w_a(x)w_b(x) - q_o w_c(x)
437        //
438        // =======================================================================
439        let step = start_timer!(|| "verify zero check");
440        // Zero check and perm check have different AuxInfo
441        let zero_check_aux_info = VPAuxInfo::<E::ScalarField> {
442            max_degree: vk.params.gate_func.degree(),
443            num_variables: num_vars,
444            phantom: PhantomData,
445        };
446        // push witness to transcript
447        for w_com in proof.witness_commits.iter() {
448            transcript.append_serializable_element(b"w", w_com)?;
449        }
450
451        let zero_check_sub_claim = <Self as ZeroCheck<E::ScalarField>>::verify(
452            &proof.zero_check_proof,
453            &zero_check_aux_info,
454            &mut transcript,
455        )?;
456
457        let zero_check_point = zero_check_sub_claim.point;
458
459        // check zero check subclaim
460        let f_eval = eval_f(&vk.params.gate_func, selector_evals, witness_gate_evals)?;
461        if f_eval != zero_check_sub_claim.expected_evaluation {
462            return Err(HyperPlonkErrors::InvalidProof(
463                "zero check evaluation failed".to_string(),
464            ));
465        }
466
467        end_timer!(step);
468        // =======================================================================
469        // 2. Verify perm_check_proof on `\{w_i(x)\}` and `permutation_oracle`
470        // =======================================================================
471        let step = start_timer!(|| "verify permutation check");
472
473        // Zero check and perm check have different AuxInfo
474        let perm_check_aux_info = VPAuxInfo::<E::ScalarField> {
475            // Prod(x) has a max degree of witnesses.len() + 1
476            max_degree: proof.witness_commits.len() + 1,
477            num_variables: num_vars,
478            phantom: PhantomData,
479        };
480        let perm_check_sub_claim = <Self as PermutationCheck<E, PCS>>::verify(
481            &proof.perm_check_proof,
482            &perm_check_aux_info,
483            &mut transcript,
484        )?;
485
486        let perm_check_point = perm_check_sub_claim
487            .product_check_sub_claim
488            .zero_check_sub_claim
489            .point;
490
491        let alpha = perm_check_sub_claim.product_check_sub_claim.alpha;
492        let (beta, gamma) = perm_check_sub_claim.challenges;
493
494        let mut id_evals = vec![];
495        for i in 0..num_witnesses {
496            let ith_point = gen_eval_point(i, log2(num_witnesses) as usize, &perm_check_point[..]);
497            id_evals.push(vk.params.eval_id_oracle(&ith_point[..])?);
498        }
499
500        // check evaluation subclaim
501        let perm_gate_eval = eval_perm_gate(
502            prod_evals,
503            frac_evals,
504            witness_perm_evals,
505            &id_evals[..],
506            perm_evals,
507            alpha,
508            beta,
509            gamma,
510            *perm_check_point.last().unwrap(),
511        )?;
512        if perm_gate_eval
513            != perm_check_sub_claim
514                .product_check_sub_claim
515                .zero_check_sub_claim
516                .expected_evaluation
517        {
518            return Err(HyperPlonkErrors::InvalidVerifier(
519                "evaluation failed".to_string(),
520            ));
521        }
522
523        end_timer!(step);
524        // =======================================================================
525        // 3. Verify the opening against the commitment
526        // =======================================================================
527        let step = start_timer!(|| "assemble commitments");
528
529        // generate evaluation points and commitments
530        let mut comms = vec![];
531        let mut points = vec![];
532
533        let perm_check_point_0 = [
534            &[E::ScalarField::zero()],
535            &perm_check_point[0..num_vars - 1],
536        ]
537        .concat();
538        let perm_check_point_1 =
539            [&[E::ScalarField::one()], &perm_check_point[0..num_vars - 1]].concat();
540        let prod_final_query_point = [
541            vec![E::ScalarField::zero()],
542            vec![E::ScalarField::one(); num_vars - 1],
543        ]
544        .concat();
545
546        // prod(x)'s points
547        comms.push(proof.perm_check_proof.prod_x_comm);
548        comms.push(proof.perm_check_proof.prod_x_comm);
549        comms.push(proof.perm_check_proof.prod_x_comm);
550        comms.push(proof.perm_check_proof.prod_x_comm);
551        points.push(perm_check_point.clone());
552        points.push(perm_check_point_0.clone());
553        points.push(perm_check_point_1.clone());
554        points.push(prod_final_query_point);
555        // frac(x)'s points
556        comms.push(proof.perm_check_proof.frac_comm);
557        comms.push(proof.perm_check_proof.frac_comm);
558        comms.push(proof.perm_check_proof.frac_comm);
559        points.push(perm_check_point.clone());
560        points.push(perm_check_point_0);
561        points.push(perm_check_point_1);
562
563        // perms' points
564        for &pcom in vk.perm_commitments.iter() {
565            comms.push(pcom);
566            points.push(perm_check_point.clone());
567        }
568
569        // witnesses' points
570        // TODO: merge points
571        for &wcom in proof.witness_commits.iter() {
572            comms.push(wcom);
573            points.push(perm_check_point.clone());
574        }
575        for &wcom in proof.witness_commits.iter() {
576            comms.push(wcom);
577            points.push(zero_check_point.clone());
578        }
579
580        // selector_poly(zero_check_point)
581        for &com in vk.selector_commitments.iter() {
582            comms.push(com);
583            points.push(zero_check_point.clone());
584        }
585
586        // - 4.4. public input consistency checks
587        //   - pi_poly(r_pi) where r_pi is sampled from transcript
588        let r_pi = transcript.get_and_append_challenge_vectors(b"r_pi", ell)?;
589
590        // check public evaluation
591        let pi_step = start_timer!(|| "check public evaluation");
592        let pi_poly = DenseMultilinearExtension::from_evaluations_slice(ell, pub_input);
593        let expect_pi_eval = evaluate_opt(&pi_poly, &r_pi[..]);
594        if expect_pi_eval != *pi_eval {
595            return Err(HyperPlonkErrors::InvalidProver(format!(
596                "Public input eval mismatch: got {}, expect {}",
597                pi_eval, expect_pi_eval,
598            )));
599        }
600        let r_pi_padded = [r_pi, vec![E::ScalarField::zero(); num_vars - ell]].concat();
601
602        comms.push(proof.witness_commits[0]);
603        points.push(r_pi_padded);
604        assert_eq!(comms.len(), proof.batch_openings.f_i_eval_at_point_i.len());
605        end_timer!(pi_step);
606        end_timer!(step);
607
608        let step = start_timer!(|| "PCS batch verify");
609        // check proof
610        let res = PCS::batch_verify(
611            &vk.pcs_param,
612            &comms,
613            &points,
614            &proof.batch_openings,
615            &mut transcript,
616        )?;
617
618        end_timer!(step);
619        end_timer!(start);
620        Ok(res)
621    }
622}

656
657    fn test_hyperplonk_helper<E: Pairing>(
658        gate_func: CustomizedGates,
659    ) -> Result<(), HyperPlonkErrors> {
660        let mut rng = test_rng();
661        let pcs_srs = MultilinearKzgPCS::<E>::gen_srs_for_testing(&mut rng, 16)?;
662
663        let num_constraints = 4;
664        let num_pub_input = 4;
665        let nv = log2(num_constraints) as usize;
666        let num_witnesses = 2;
667
668        // generate index
669        let params = HyperPlonkParams {
670            num_constraints,
671            num_pub_input,
672            gate_func,
673        };
674        let permutation = identity_permutation(nv, num_witnesses);
675        let q1 = SelectorColumn(vec![
676            E::ScalarField::one(),
677            E::ScalarField::one(),
678            E::ScalarField::one(),
679            E::ScalarField::one(),
680        ]);
681        let index = HyperPlonkIndex {
682            params,
683            permutation,
684            selectors: vec![q1],
685        };
686
687        // generate pk and vks
688        let (pk, vk) =
689            <PolyIOP<E::ScalarField> as HyperPlonkSNARK<E, MultilinearKzgPCS<E>>>::preprocess(
690                &index, &pcs_srs,
691            )?;
692
693        // w1 := [0, 1, 2, 3]
694        let w1 = WitnessColumn(vec![
695            E::ScalarField::zero(),
696            E::ScalarField::one(),
697            E::ScalarField::from(2u128),
698            E::ScalarField::from(3u128),
699        ]);
700        // w2 := [0^5, 1^5, 2^5, 3^5]
701        let w2 = WitnessColumn(vec![
702            E::ScalarField::zero(),
703            E::ScalarField::one(),
704            E::ScalarField::from(32u128),
705            E::ScalarField::from(243u128),
706        ]);
707        // public input = w1
708        let pi = w1.clone();
709
710        // generate a proof and verify
711        let proof = <PolyIOP<E::ScalarField> as HyperPlonkSNARK<E, MultilinearKzgPCS<E>>>::prove(
712            &pk,
713            &pi.0,
714            &[w1.clone(), w2.clone()],
715        )?;
716
717        let _verify =
718            <PolyIOP<E::ScalarField> as HyperPlonkSNARK<E, MultilinearKzgPCS<E>>>::verify(
719                &vk, &pi.0, &proof,
720            )?;
721
722        // bad path 1: wrong permutation
723        let rand_perm: Vec<E::ScalarField> = random_permutation(nv, num_witnesses, &mut rng);
724        let mut bad_index = index;
725        bad_index.permutation = rand_perm;
726        // generate pk and vks
727        let (_, bad_vk) =
728            <PolyIOP<E::ScalarField> as HyperPlonkSNARK<E, MultilinearKzgPCS<E>>>::preprocess(
729                &bad_index, &pcs_srs,
730            )?;
731        assert!(!<PolyIOP<E::ScalarField> as HyperPlonkSNARK<
732            E,
733            MultilinearKzgPCS<E>,
734        >>::verify(&bad_vk, &pi.0, &proof,)?);
735
736        // bad path 2: wrong witness
737        let mut w1_bad = w1;
738        w1_bad.0[0] = E::ScalarField::one();
739        assert!(
740            <PolyIOP<E::ScalarField> as HyperPlonkSNARK<E, MultilinearKzgPCS<E>>>::prove(
741                &pk,
742                &pi.0,
743                &[w1_bad, w2],
744            )
745            .is_err()
746        );
747
748        Ok(())
749    }
