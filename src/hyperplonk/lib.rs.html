24/// A trait for HyperPlonk SNARKs.
25/// A HyperPlonk is derived from ZeroChecks and PermutationChecks.
26pub trait HyperPlonkSNARK<E, PCS>: PermutationCheck<E, PCS>
27where
28    E: Pairing,
29    PCS: PolynomialCommitmentScheme<E>,
30{
31    type Index;
32    type ProvingKey;
33    type VerifyingKey;
34    type Proof;
35
36    /// Generate the preprocessed polynomials output by the indexer.
37    ///
38    /// Inputs:
39    /// - `index`: HyperPlonk index
40    /// - `pcs_srs`: Polynomial commitment structured reference string
41    ///
42    /// Outputs:
43    /// - The HyperPlonk proving key, which includes the preprocessed
44    ///   polynomials.
45    /// - The HyperPlonk verifying key, which includes the preprocessed
46    ///   polynomial commitments
47    fn preprocess(
48        index: &Self::Index,
49        pcs_srs: &PCS::SRS,
50    ) -> Result<(Self::ProvingKey, Self::VerifyingKey), HyperPlonkErrors>;
51
52    /// Generate HyperPlonk SNARK proof.
53    ///
54    /// Inputs:
55    /// - `pk`: circuit proving key
56    /// - `pub_input`: online public input
57    /// - `witness`: witness assignment
58    ///
59    /// Outputs:
60    /// - The HyperPlonk SNARK proof.
61    fn prove(
62        pk: &Self::ProvingKey,
63        pub_input: &[E::ScalarField],
64        witnesses: &[WitnessColumn<E::ScalarField>],
65    ) -> Result<Self::Proof, HyperPlonkErrors>;
66
67    /// Verify the HyperPlonk proof.
68    ///
69    /// Inputs:
70    /// - `vk`: verifying key
71    /// - `pub_input`: online public input
72    /// - `proof`: HyperPlonk SNARK proof challenges
73    ///
74    /// Outputs:
75    /// - Return a boolean on whether the verification is successful
76    fn verify(
77        vk: &Self::VerifyingKey,
78        pub_input: &[E::ScalarField],
79        proof: &Self::Proof,
80    ) -> Result<bool, HyperPlonkErrors>;
81}
